To package this into a clean executable for end users without them needing to install Node.js or run terminal commands, you can use Pkg (or Electron-Builder with your local server approach).

Packaging a local Express server + an auto-launcher into a single .exe file gives users a double-clickable app experience.

Step 1: Install pkg
Install pkg as a development dependency:

Bash
npm install -D pkg
Step 2: Configure package.json
Add a pkg section to your package.json to include your dist folder and static assets in the final binary bundle:

JSON
{
  "name": "tagflix",
  "version": "1.0.0",
  "main": "server.cjs",
  "scripts": {
    "build": "vite build",
    "package": "pkg . --targets node18-win-x64,node18-macos-x64,node18-linux-x64 --out-path dist-app"
  },
  "pkg": {
    "assets": [
      "dist/**/*"
    ],
    "targets": [
      "node18-win-x64"
    ],
    "outputPath": "dist-app"
  }
}
Step 3: Adjust server.cjs for Executable Paths
When your server runs inside a compiled pkg executable, standard relative paths like __dirname point inside a virtual filesystem (/snapshot/...). Adjust the pathing logic slightly in your server.cjs:

JavaScript
const express = require('express');
const path = require('path');
const fs = require('fs');
const open = require('open');
const { execSync } = require('child_process');

const app = express();
const PORT = 5173;

// Resolve path properly whether running via Node or as a compiled pkg binary
const isPkg = typeof process.pkg !== 'undefined';
const baseDir = isPkg ? path.dirname(process.execPath) : __dirname;
const distDir = isPkg 
  ? path.join(__dirname, 'dist') // Embed inside pkg virtual snapshot
  : path.join(__dirname, 'dist');

app.use(express.static(distDir));

app.get('*', (req, res) => {
  const indexFile = path.join(distDir, 'index.html');
  res.sendFile(indexFile);
});

function getPreferredAppBrowser() {
  const platform = process.platform;

  if (platform === 'win32') {
    const edgePaths = [
      process.env['PROGRAMFILES(X86)'] + '\\Microsoft\\Edge\\Application\\msedge.exe',
      process.env['PROGRAMFILES'] + '\\Microsoft\\Edge\\Application\\msedge.exe',
    ];
    if (edgePaths.some(p => fs.existsSync(p))) return 'msedge';

    const chromePaths = [
      process.env['PROGRAMFILES(X86)'] + '\\Google\\Chrome\\Application\\chrome.exe',
      process.env['PROGRAMFILES'] + '\\Google\\Chrome\\Application\\chrome.exe',
      process.env['LOCALAPPDATA'] + '\\Google\\Chrome\\Application\\chrome.exe',
    ];
    if (chromePaths.some(p => fs.existsSync(p))) return 'google chrome';
  } else if (platform === 'darwin') {
    if (fs.existsSync('/Applications/Microsoft Edge.app')) return 'microsoft edge';
    if (fs.existsSync('/Applications/Google Chrome.app')) return 'google chrome';
  }

  return null;
}

app.listen(PORT, '127.0.0.1', async () => {
  const targetUrl = `http://127.0.0.1:${PORT}`;
  console.log(`[tagflix] App active at ${targetUrl}`);

  const detectedBrowser = getPreferredAppBrowser();

  try {
    if (detectedBrowser) {
      await open(targetUrl, {
        app: {
          name: open.apps[detectedBrowser] || detectedBrowser,
          arguments: [`--app=${targetUrl}`],
        },
      });
    } else {
      await open(targetUrl);
    }
  } catch (err) {
    console.error('Browser launch error:', err);
  }
});
Step 4: Build and Distribute
Run the following commands:

Build your frontend production bundle:

Bash
npm run build
Package the app into a standalone binary:

Bash
npx pkg .
You will get a single standalone executable file (e.g., tagflix.exe in the dist-app folder).

End User Experience
No Prerequisites Required: The user does not need Node.js, npm, or any code dependencies installed.

One-Click Launch: The user simply double-clicks tagflix.exe.

Seamless Experience: It boots the embedded local server silently, detects Edge/Chrome on their computer, and opens your app directly in frameless App Mode without CORS or stream playback 500/502 issues.
